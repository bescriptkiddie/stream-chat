// Vite 开发原理（ESM）
import { foo } from './foo.js';  // 浏览器直接请求 foo.js
import { bar } from './bar.js';  // 浏览器直接请求 bar.js
// ✅ 按需加载，修改 foo.js 只更新 foo.js

// Webpack 开发原理（Bundle）
// 1. 入口文件 -> 依赖分析 -> 打包成 bundle.js
// 2. 修改任何文件 -> 重新打包整个 bundle
// ❌ 全量打包，速度随项目增大而变慢

🎤 面试高频 QA

Q1: Vite 为什么比 Webpack 快？

核心原因：

  1. 1. 开发模式：
    • Vite: 基于 ESM，浏览器按需加载模块，无需打包
    • Webpack: 全量打包成 bundle，启动慢
  2. 2. 热更新（HMR）：
    • Vite: 只更新修改的模块
    • Webpack: 重新打包相关模块
  3. 3. 依赖预构建：
    • Vite: esbuild 预构建依赖（Go 编写，速度快 10-100x）
    • Webpack: JavaScript 实现，速度慢
Q2: 什么是 Tree Shaking？如何生效？

📚 定义：

基于 ES Module 的静态分析，删除未使用的代码（Dead Code Elimination）

✅ 生效条件：

  1. 使用 ES Module（import/export）
  2. 代码无副作用（或标记 sideEffects: false）
  3. 生产环境构建
  4. 使用支持 Tree Shaking 的打包工具
Q3: 如何分析打包体积？
// 1. rollup-plugin-visualizer
npm install --save-dev rollup-plugin-visualizer

// vite.config.js
import { visualizer } from 'rollup-plugin-visualizer';
export default {
  plugins: [
    visualizer({
      open: true, // 自动打开
      gzipSize: true,
      brotliSize: true,
    })
  ]
}

// 2. vite-bundle-analyzer  
npm install --save-dev vite-bundle-analyzer

// 3. 命令行查看
npm run build -- --report

// 4. webpack-bundle-analyzer (Webpack)
npx webpack-bundle-analyzer dist/stats.json
Q4: 代码分割有哪些方式？
  1. 路由懒加载：React.lazy() + Suspense
  2. 组件懒加载：按需加载重型组件
  3. 第三方库按需引入：lodash-es
  4. 动态 import()：条件加载
  5. 手动分包：manualChunks 配置
Q5: 生产环境如何优化首屏加载？

🚀 综合优化方案：

  1. 代码层面：
    • 路由懒加载
    • Tree Shaking
    • 删除 console.log
  2. 资源层面：
    • 图片压缩/WebP 格式
    • 字体子集化
    • CDN 加速
  3. 网络层面：
    • HTTP/2 多路复用
    • Gzip/Brotli 压缩
    • 浏览器缓存（Cache-Control）
  4. 渲染层面：
    • SSR/SSG（Next.js）
    • 骨架屏/Loading
    • 关键 CSS 内联

✅ Vite 最佳实践

✅ 应该做的

  • • 使用 ESM 模块（import/export）
  • • 配置依赖预构建（optimizeDeps）
  • • 手动分包（manualChunks）
  • • 使用 visualizer 分析体积
  • • 路由懒加载 + Suspense
  • • 按需引入第三方库
  • • 生产环境删除 console

❌ 不应该做的

  • • 全量引入 lodash（用 lodash-es）
  • • 全量引入 moment（用 dayjs）
  • • 使用 CommonJS（module.exports）
  • • 忽略 chunk 大小警告
  • • 所有代码打包成一个文件
  • • 不分析打包产物
  • • 开发环境使用生产构建

⚠️ 常见陷阱

❌ 陷阱 1：ESM 和 CJS 混用

// ❌ 混用导致 Tree Shaking 失效
import foo from 'esm-lib';
const bar = require('cjs-lib'); // CommonJS

// ✅ 统一使用 ESM
import foo from 'esm-lib';
import bar from 'cjs-lib';

❌ 陷阱 2：动态 import 路径

// ❌ 完全动态路径无法分析
const lang = 'zh-CN';
import(`./locales/${lang}.js`); // Vite 无法静态分析

// ✅ 使用有限的动态路径
import(`./locales/${lang}.js`); // 确保 ./locales 下所有文件都存在

// ✅ 或使用预定义映射
const locales = {
  'zh-CN': () => import('./locales/zh-CN.js'),
  'en-US': () => import('./locales/en-US.js')
};
locales[lang]();

❌ 陷阱 3：忘记配置 sideEffects

// package.json
{
  "name": "my-lib",
  "sideEffects": false  // ✅ 启用 Tree Shaking
}

// ❌ 如果有副作用但标记为 false，可能导致代码被错误删除
import './polyfill.js';  // 这个会被删除！
