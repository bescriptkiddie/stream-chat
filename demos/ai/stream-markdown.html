const parseMarkdown = (text) => {
  let html = text;

  // 1. 代码块（优先处理，避免内部语法被解析）
  html = html.replace(/```(\w+)?\n([\s\S]*?)```/g, (match, lang, code) => {
    return `<pre><code class="language-${lang}">${escapeHtml(code)}</code></pre>`;
  });

  // 2. 行内代码
  html = html.replace(/`([^`]+)`/g, '<code>$1</code>');

  // 3. 标题 (H1-H3)
  html = html.replace(/^### (.*$)/gim, '<h3>$1</h3>');
  html = html.replace(/^## (.*$)/gim, '<h2>$1</h2>');
  html = html.replace(/^# (.*$)/gim, '<h1>$1</h1>');

  // 4. 粗体、斜体
  html = html.replace(/\*\*([^*]+)\*\*/g, '<strong>$1</strong>');
  html = html.replace(/\*([^*]+)\*/g, '<em>$1</em>');

  // 5. 链接
  html = html.replace(/\[([^\]]+)\]\(([^)]+)\)/g, '<a href="$2">$1</a>');

  // 6. 列表
  html = html.replace(/^[\s]*[-*] (.+)$/gim, '<li>$1</li>');
  html = html.replace(/(<li>.*<\/li>)/s, '<ul>$1</ul>');

  return html;
};

  • 使用虚拟滚动处理长文档
  • 节流更新频率（100-200ms 批量刷新）

  • 代码块跨 chunk：缓冲未闭合的 ```

4. 安全性
  • 使用 DOMPurify 清理 HTML
  • 限制允许的标签和属性
  • CSP 策略防止 XSS

👔
面试官：流式 Markdown 渲染的核心难点是什么？
✅ 好的回答：

三个核心难点：

  1. 边界处理：Markdown 语法可能跨多个 chunk
    chunk1: "```javascri"
chunk2: "pt\ncode\n```"
// 需要缓冲，等待完整的代码块
  2. 性能优化：每次追加都重新解析，CPU 占用高

    解决：增量解析 + 节流更新 + React.memo

  3. 视觉连贯性：避免布局跳动和闪烁

    解决：预留空间 + 骨架屏 + CSS transition

👔
面试官：如何处理代码块跨 chunk 的情况？
✅ 实现思路：

状态机方案：

const [parseState, setParseState] = useState({
  inCodeBlock: false,
  codeBlockLang: '',
  codeBlockContent: '',
  buffer: ''
});

function processChunk(chunk) {
  let text = parseState.buffer + chunk;
  
  // 检查是否开始代码块
  if (text.includes('```')) {
    const match = text.match(/```(\w+)?\n/);
    if (match) {
      setParseState({
        ...parseState,
        inCodeBlock: true,
        codeBlockLang: match[1] || 'plaintext'
      });
    }
  }
  
  // 检查是否结束代码块
  if (parseState.inCodeBlock && text.includes('```')) {
    setParseState({
      ...parseState,
      inCodeBlock: false,
      buffer: ''
    });
  } else {
    // 保存到缓冲区
    setParseState({
      ...parseState,
      buffer: text
    });
  }
}

💡 关键：维护解析状态，缓冲不完整的语法块

面试官：如何优化大段文本的渲染性能？
✅ 多层优化方案：

1. 节流更新（减少渲染次数）

let buffer = '';
const flushInterval = setInterval(() => {
  if (buffer) {
    setStreamedText(prev => prev + buffer);
    buffer = '';
  }
}, 100); // 每 100ms 批量更新

<VirtualList

3. Web Worker 解析（不阻塞主线程）

const worker = new Worker('/markdown-worker.js');
worker.postMessage({ text: streamedText });
worker.onmessage = (e) => {
  setRenderedHtml(e.data.html);
};

4. React.memo 缓存（避免重复渲染）

const MarkdownBlock = React.memo(({ content }) => {
  return <ReactMarkdown>{content}</ReactMarkdown>;
});

❓ 面试高频 QA

Q1: React-Markdown 和手写解析器各有什么优缺点？

React-Markdown

优点：

  • 功能完整，支持 GFM
  • 插件生态丰富
  • 安全性好（XSS 防护）
  • 维护成本低

缺点：

  • 包体积较大（~50KB）
  • 性能开销较高
  • 定制化受限

手写解析器

优点：

  • 体积小，性能好
  • 完全可控
  • 针对性优化
  • 按需实现功能

缺点：

  • 开发成本高
  • 功能不完整
  • 安全性需自行保证
  • 维护成本高

💡 建议：生产环境用成熟库，面试/Demo 可手写展示理解

Q2: 如何防止 Markdown 中的 XSS 攻击？

多层防护：

  1. 使用 DOMPurify 清理 HTML：
    import DOMPurify from 'dompurify';

const cleanHtml = DOMPurify.sanitize(html, {
  ALLOWED_TAGS: ['p', 'a', 'strong', 'em', 'code', 'pre'],
  ALLOWED_ATTR: ['href', 'target']
});
  2. 禁用危险的 Markdown 特性：
    <ReactMarkdown
  disallowedElements={['script', 'iframe', 'object']}
  unwrapDisallowed={true}
>
  {text}
</ReactMarkdown>
  3. 设置 CSP 策略：
    <meta http-equiv="Content-Security-Policy" 
      content="default-src 'self'; script-src 'none';">

使用 KaTeX 或 MathJax：

Q4: 如何实现 Markdown 编辑器的实时预览？

双栏同步滚动方案：

const MarkdownEditor = () => {
  const [text, setText] = useState('');
  const editorRef = useRef();
  const previewRef = useRef();

  const handleScroll = (e) => {
    const { scrollTop, scrollHeight, clientHeight } = e.target;
    const scrollPercent = scrollTop / (scrollHeight - clientHeight);
    
    // 同步滚动预览区
    const preview = previewRef.current;
    const targetScroll = scrollPercent * 
      (preview.scrollHeight - preview.clientHeight);
    preview.scrollTop = targetScroll;
  };

  return (
    <div className="grid grid-cols-2">
      <textarea
        ref={editorRef}
        value={text}
        onChange={(e) => setText(e.target.value)}
        onScroll={handleScroll}
      />
      <div ref={previewRef}>
        <ReactMarkdown>{text}</ReactMarkdown>
      </div>
    </div>
  );
};
